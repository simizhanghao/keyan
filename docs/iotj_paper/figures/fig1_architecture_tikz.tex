% ----- Stage 1: Tokenization -----
\node[input] (iq) {IQ/FFT/AP\\views};
\node[main, right=of iq] (cnn) {CNN\\stem};
\node[main, right=of cnn] (tm) {Main tokens\\$T_m$};

\node[input, below=0.92cm of iq] (oobspec) {OOB\\spectrum};
\node[oob, right=of oobspec] (oobtok) {OOB\\tokenizer};
\node[oob, right=of oobtok] (to) {OOB tokens\\$T_o$};

\draw[arrow] (iq) -- (cnn);
\draw[arrow] (cnn) -- (tm);
\draw[arrow] (oobspec) -- (oobtok);
\draw[arrow] (oobtok) -- (to);

\node[note, below=0.08cm of tm] (shape1) {$T_m,T_o\in\mathbb{R}^{P\times D}$\\$P=32,\;D=64$};

% ----- Stage 2: Fusion -----
\node[main, right=1.05cm of tm] (pos) {Pos./chirp\\embedding};
\node[fuse, right=of pos, minimum width=1.52cm] (attn) {OOB-guided\\cross-attn};
\node[fuse, right=of attn, minimum width=1.42cm] (gate) {Gated residual\\injection};
\node[fuse, right=of gate] (tf) {Fused tokens\\$T_f$};

\draw[arrow] (tm) -- (pos);
\draw[arrow] (pos) -- node[above, note, pos=0.55] {$Q{=}T_m$} (attn);
\draw[arrow] (to.east) -- ++(0.42,0) |- node[pos=0.24, below, note] {$K,V{=}T_o$} (attn.south);
\draw[arrow] (attn) -- (gate);
\draw[arrow] (gate) -- (tf);

% ----- Stage 3: Authentication -----
\node[enc, right=1.0cm of tf, minimum width=1.42cm] (hstu) {RF-HSTU\\encoder\\[-1pt]\scriptsize gated mixing};
\node[out, right=of hstu] (logits) {Window logits\\$s^{(k)}$};
\node[out, right=of logits, minimum width=1.50cm] (vote) {Mean over\\$K$ windows};
\node[out, right=of vote] (decision) {File-level\\decision\\$\hat{y}$};

\draw[arrow] (tf) -- (hstu);
\draw[arrow] (hstu) -- (logits);
\draw[arrow] (logits) -- (vote);
\draw[arrow] (vote) -- (decision);
