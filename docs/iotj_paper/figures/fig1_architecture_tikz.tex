% ----- Stage 1: Tokenization -----
\node[inputbox] (iq) {IQ/FFT/AP\\views};
\node[mainbox, right=of iq] (cnn) {CNN\\stem};
\node[mainbox, right=of cnn] (tm) {Main tokens\\$T_m$};

\node[inputbox, below=0.92cm of iq] (oobspec) {OOB\\spectrum};
\node[oobbox, right=of oobspec] (oobtok) {OOB\\tokenizer};
\node[oobbox, right=of oobtok] (oobtokens) {OOB tokens\\$T_o$};

\draw[arr] (iq) -- (cnn);
\draw[arr] (cnn) -- (tm);
\draw[arr] (oobspec) -- (oobtok);
\draw[arr] (oobtok) -- (oobtokens);

\node[note, below=0.08cm of tm] (shape1) {$T_m,T_o\in\mathbb{R}^{P\times D}$\\$P=32,\;D=64$};

% ----- Stage 2: Fusion -----
\node[mainbox, right=1.05cm of tm] (pos) {Pos./chirp\\embedding};
\node[fusebox, right=of pos, minimum width=1.52cm] (attn) {OOB-guided\\cross-attn};
\node[fusebox, right=of attn, minimum width=1.42cm] (gate) {Gated residual\\injection};
\node[fusebox, right=of gate] (tf) {Fused tokens\\$T_f$};

\draw[arr] (tm) -- (pos);
\draw[arr] (pos) -- node[above, note, pos=0.55] {$Q{=}T_m$} (attn);
\draw[arr] (oobtokens.east) -- ++(0.42,0) |- node[pos=0.24, below, note] {$K,V{=}T_o$} (attn.south);
\draw[arr] (attn) -- (gate);
\draw[arr] (gate) -- (tf);

% ----- Stage 3: Authentication -----
\node[encbox, right=1.0cm of tf, minimum width=1.42cm] (hstu) {RF-HSTU\\encoder\\[-1pt]{\footnotesize gated mixing}};
\node[outbox, right=of hstu] (logits) {Window logits\\$s^{(k)}$};
\node[outbox, right=of logits, minimum width=1.50cm] (vote) {Mean over\\$K$ windows};
\node[outbox, right=of vote] (decision) {File-level\\decision\\$\hat{y}$};

\draw[arr] (tf) -- (hstu);
\draw[arr] (hstu) -- (logits);
\draw[arr] (logits) -- (vote);
\draw[arr] (vote) -- (decision);
